Confige the openshift-logging loki-stack / Log Forwarder

[source, bash]
----
cat << 'EOF' > 1_bucketname.yaml
apiVersion: objectbucket.io/v1alpha1
kind: ObjectBucketClaim
metadata:
  name: loki-s3
  namespace: openshift-logging
spec:
  generateBucketName: loki-s3
  storageClassName: ocs-storagecluster-ceph-rgw
EOF
cat << 'EOF' > 2_secret.sh
#!/bin/bash
BUCKET_HOST=$(oc get -n openshift-logging configmap loki-s3 -o jsonpath='{.data.BUCKET_HOST}')
BUCKET_NAME=$(oc get -n openshift-logging configmap loki-s3 -o jsonpath='{.data.BUCKET_NAME}')
BUCKET_PORT=$(oc get -n openshift-logging configmap loki-s3 -o jsonpath='{.data.BUCKET_PORT}')
ACCESS_KEY_ID=$(oc get -n openshift-logging secret loki-s3 -o jsonpath='{.data.AWS_ACCESS_KEY_ID}' | base64 -d)
SECRET_ACCESS_KEY=$(oc get -n openshift-logging secret loki-s3 -o jsonpath='{.data.AWS_SECRET_ACCESS_KEY}' | base64 -d)

echo "BUCKET_HOST=$BUCKET_HOST"
echo "BUCKET_NAME=$BUCKET_NAME"
echo "BUCKET_PORT=$BUCKET_PORT"
echo "ACCESS_KEY_ID=$ACCESS_KEY_ID"
echo "SECRET_ACCESS_KEY=$SECRET_ACCESS_KEY"

oc create -n openshift-logging secret generic logging-loki-s3 --from-literal=access_key_id=$ACCESS_KEY_ID --from-literal=access_key_secret=$SECRET_ACCESS_KEY --from-literal=bucketnames=$BUCKET_NAME --from-literal=endpoint="https://$BUCKET_HOST:$BUCKET_PORT"
EOF
cat << 'EOF' > 3_loki-stack.yaml
apiVersion: loki.grafana.com/v1
kind: LokiStack
metadata:
  name: logging-loki
  namespace: openshift-logging
spec:
  managementState: Managed
  size: 1x.extra-small
  hashRing:
    type: memberlist
    memberlist:
      instanceAddrType: podIP
  storage:
    schemas:
      - version: v13
        effectiveDate: "2020-10-24"
    secret:
      name: logging-loki-s3
      type: s3
    tls:
      caName: loki-s3-ca
  storageClassName: csi-cephrbd
  tenants:
    mode: openshift-logging
EOF
cat << 'EOF' > 4_serviceaccount.yaml
apiVersion: v1
kind: ServiceAccount
metadata:
  name: log-forwarder
  namespace: openshift-logging
EOF
cat << 'EOF' > 5_cluster-role-binding.yaml
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: collect-application-logs-log-forwarder
roleRef:
  apiGroup: rbac.authorization.k8s.io
  kind: ClusterRole
  name: collect-application-logs
subjects:
- kind: ServiceAccount
  name: log-forwarder
  namespace: openshift-logging
---
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: collect-audit-logs-log-forwarder
roleRef:
  apiGroup: rbac.authorization.k8s.io
  kind: ClusterRole
  name: collect-audit-logs
subjects:
- kind: ServiceAccount
  name: log-forwarder
  namespace: openshift-logging
---
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: collect-infrastructure-logs-log-forwarder
roleRef:
  apiGroup: rbac.authorization.k8s.io
  kind: ClusterRole
  name: collect-infrastructure-logs
subjects:
- kind: ServiceAccount
  name: log-forwarder
  namespace: openshift-logging
EOF
cat << 'EOF' > 6_clusterLogForwarder.yaml
apiVersion: observability.openshift.io/v1
kind: ClusterLogForwarder
metadata:
  name: instance
  namespace: openshift-logging
spec:
  serviceAccount:
    name: log-forwarder
  outputs:
    - name: elasticsearch-secure
      type: syslog
      syslog:
        url: udp://10.215.1.154:514
        rfc: RFC5424
  pipelines:
    - name: send-audit-logs
      inputRefs:
        - audit
      outputRefs:
        - elasticsearch-secure
    - name: send-app-logs
      inputRefs:
        - application
      outputRefs:
        - elasticsearch-secure
    - name: send-infra-logs
      inputRefs:
        - infrastructure
      outputRefs:
        - elasticsearch-secure
EOF
cat << 'EOF' > 7_cluster_logging.yaml
apiVersion: logging.openshift.io/v1
kind: ClusterLogging
metadata:
  name: instance
  namespace: openshift-logging
spec:
  collection:
    type: vector
  logStore:
    elasticsearch:
      resources:
        requests:
          memory: 2Gi
      storage:
        storageClassName: csi-cephrbd
        size: 200G
      nodeSelector:
        node-role.kubernetes.io/infra: ""
      nodeCount: 3
      redundancyPolicy: SingleRedundancy
    retentionPolicy:
      infra:
        maxAge: 3d
      audit:
        maxAge: 3d
    type: elasticsearch
  visualization:
    kibana:
      nodeSelector:
        node-role.kubernetes.io/infra: ""
      replicas: 1
    nodeSelector:
        node-role.kubernetes.io/infra: ""
    type: kibana
  managementState: Managed
EOF
----

----
Suggested Execution Path:
Apply the OBC: oc apply -f 1_bucketname.yaml

Wait for the bucket to be provisioned (Check with oc get obc -n openshift-logging).

Run the script: ./2_secret.sh to generate the Loki secret.

Deploy Loki: oc apply -f 3_loki-stack.yaml

Setup Permissions: oc apply -f 4_serviceaccount.yaml -f 5_cluster-role-binding.yaml

Deploy Logging & Forwarding: oc apply -f 6_clusterLogForwarder.yaml -f 7_cluster_logging.yaml
----

----
